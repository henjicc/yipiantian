cmake_minimum_required(VERSION 3.20)
project(FarmDesktop LANGUAGES CXX)
add_executable(FarmDesktop WIN32 main.cpp)
target_compile_features(FarmDesktop PRIVATE cxx_std_17)
target_compile_definitions(FarmDesktop PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(FarmDesktop PRIVATE user32 shell32 dwmapi gdi32 wtsapi32 oleacc ole32 oleaut32 uuid)
if(MSVC)
  set_property(TARGET FarmDesktop PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(FarmDesktop PRIVATE /W4 /utf-8)
else()
  target_compile_options(FarmDesktop PRIVATE -Wall -Wextra)
  target_link_options(FarmDesktop PRIVATE -municode -static -static-libgcc -static-libstdc++)
endif()
